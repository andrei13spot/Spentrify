import { Ionicons } from "@expo/vector-icons";
import { router, useLocalSearchParams } from "expo-router";
import React, { useMemo, useState } from "react";
import {
    Share,
    StyleSheet,
    Text,
    TouchableOpacity,
    View,
} from "react-native";

type InsightSentiment = "positive" | "negative" | "neutral";

type InsightPayload = {
  id: string;
  sentiment: InsightSentiment;
  summary: string;
  actionPoints: string;
  recommendations: string;
};

const SPENTRIFY_BLUE = "#3B82F6";

/* TEMP MOCK DATA (AI/BACKEND LATER) */
const MOCK_INSIGHTS: Record<string, InsightPayload> = {
  "weekly-trend": {
    id: "weekly-trend",

    actionPoints:

    recommendations:

};

type TabKey = "summary" | "actionPoints" | "recommendations";

export default function InsightDetailScreen() {
  const { id } = useLocalSearchParams<{ id: string }>();
  const [activeTab, setActiveTab] = useState<TabKey>("summary");

  const insight = useMemo(() => {
    return typeof id === "string" ? MOCK_INSIGHTS[id] : null;
  }, [id]);

  const tabContent = useMemo(() => {
    if (!insight) return "No data available.";
    if (activeTab === "summary") return insight.summary;
    if (activeTab === "actionPoints") return insight.actionPoints;
    return insight.recommendations;
  }, [activeTab, insight]);

  const handleShare = async () => {
    try {
      await Share.share({
        message: tabContent,
      });
    } catch {}
  };

  return (
    <SafeAreaView style={styles.safeArea} edges={["top"]}>
      <View style={styles.container}>
        {/* HEADER — MATCHES NOTIFICATIONS */}
        <View style={styles.header}>
          <TouchableOpacity onPress={() => router.back()}>
            <Ionicons name="chevron-back" size={24} color="#111827" />
          </TouchableOpacity>

          <Text style={styles.headerTitle}>Insight</Text>

          {/* SETTINGS → MORE TAB (FIXED) */}
          <TouchableOpacity onPress={() => router.push("/(tabs)/more")}>
            <Ionicons name="settings-outline" size={22} color="#111827" />
          </TouchableOpacity>
        </View>

        {/* EMOJI / ICON */}
        <View style={styles.emojiCard}>
          <Text style={styles.emoji}>🧠</Text>
        </View>

        {/* TABS */}
        <View style={styles.tabs}>
          <Tab label="Summary" active={activeTab === "summary"} onPress={() => setActiveTab("summary")} />
          <Tab label="Action Points" active={activeTab === "actionPoints"} onPress={() => setActiveTab("actionPoints")} />
          <Tab label="Recommendations" active={activeTab === "recommendations"} onPress={() => setActiveTab("recommendations")} />
        </View>

        {/* CONTENT */}
        <View style={styles.contentCard}>
          <Text style={styles.contentText}>{tabContent}</Text>
        </View>

        {/* SHARE */}
        <TouchableOpacity style={styles.shareBtn} onPress={handleShare}>
          <Text style={styles.shareText}>Share Result</Text>
          <Ionicons name="share-outline" size={18} color="#fff" />
        </TouchableOpacity>
      </View>
    </SafeAreaView>
  );
}

/* TAB BUTTON */
function Tab({
  label,
  active,
  onPress,
}: {
  label: string;
  active: boolean;
  onPress: () => void;
}) {
  return (
    <TouchableOpacity
      onPress={onPress}
      style={[styles.tabBtn, active && styles.tabBtnActive]}
    >
      <Text style={[styles.tabText, active && styles.tabTextActive]}>
        {label}
      </Text>
    </TouchableOpacity>
  );
}
